11	Эффективная нагрузка	=CHOOSE(regime_no,'05_Расчёт'!B17,'05_Расчёт'!C17,'05_Расчёт'!D17,'05_Расчёт'!E17,'05_Расчёт'!F17,'05_Расчёт'!G17,'05_Расчёт'!H17)  ⟹ None
12	Точка безубыточности, ₽/год	=CHOOSE(regime_no,'05_Расчёт'!B22,'05_Расчёт'!C22,'05_Расчёт'!D22,'05_Расчёт'!E22,'05_Расчёт'!F22,'05_Расчёт'!G22,'05_Расчёт'!H22)  ⟹ None
13	ВРЕМЯ — иерархия
14	ЧИСТОЕ время (всё доступное)	=net_time  ⟹ None	производств. календарь − отпуск − больничный − час/день
15	  ├ ОПЕРАЦИОННОЕ (второстепенное)	=side_time  ⟹ None	продвижение + учёт
16	  ├ РЕЗЕРВ на форс-мажоры	=fm_time  ⟹ None	переносы, отмены, срывы
17	  └ ПРОЕКТНОЕ за год	=project_pool  ⟹ None	на выполнение всех проектов
18	      ├ ПРОФИЛЬНОЕ (по профилю)	=core_time  ⟹ None	S × (1+коэф.): площадка + обработка
19	      │   ├ СЪЁМОЧНОЕ ← ЦЕЛЬ	=shooting_time  ⟹ None	S — работа на площадке
20	      │   └ постпродакшн	=post_time  ⟹ None	S × коэффициент — обработка
21	      └ работа с клиентом	=client_total  ⟹ None	1 ч на проект: продажа, согласования
22	Продолжительность съёмки (S)	=shoot_duration  ⟹ None	типовой заказ, задаёт пользователь
23	Коэффициент обработки	=post_ratio  ⟹ None	часов обработки на 1 час съёмки
24	Проектное время на 1 проект	=project_time  ⟹ None	S × (1+коэф.) + 1
25	Проектов в год	=projects_year  ⟹ None
26	Проектов в неделю	=projects_week  ⟹ None	при авторежиме = 1
28	ПРОВЕРКА ИЕРАРХИИ ВРЕМЕНИ
29	Операционное + резерв + проектное = чистое	=IF(ABS(side_time+fm_time+project_pool-net_time)<0.01,"OK","ОШИБКА")  ⟹ None
30	Профильное + клиентское = проектное	=IF(ABS(core_time+client_total-project_pool)<0.01,"OK","ОШИБКА")  ⟹ None
31	Съёмочное + постпродакшн = профильное	=IF(ABS(shooting_time+post_time-core_time)<0.01,"OK","ОШИБКА")  ⟹ None

48	Точки безубыточности сходятся в ноль	=IF(SUMPRODUCT(--(ABS('05_Расчёт'!B23:H23)>1))=0,"OK","ОШИБКА")  ⟹ None

53	Бухгалтер: либо деньги, либо время	=IF(OR(AND(var_accounting>0,accounting_time=0),AND(var_accounting=0,accounting_time>0)),"OK","ПРОВЕРЬТЕ")  ⟹ None

57	Сайт: либо сумма, либо часы	=IF(OR(AND(dep_website>0,site_divisor=0),AND(dep_website=0,site_divisor>=0)),"OK","ПРОВЕРЬТЕ")  ⟹ None
